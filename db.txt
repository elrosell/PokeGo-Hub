CREATE DATABASE poke_rodrigo;
USE poke_rodrigo;

-- Tabla base para equipos por región
CREATE TABLE equipos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    region VARCHAR(50),
    tipo_equipo VARCHAR(50),
    pokemon_nombres VARCHAR(255),
    autor VARCHAR(80) DEFAULT 'Delgado Cerros Rodrigo Daniel',
    notas TEXT
);

-- Catálogo de Pokémon (para IVs, counters, evoluciones)
CREATE TABLE pokemon (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(60) NOT NULL,
    base_attack INT,
    base_defense INT,
    base_stamina INT,
    generation VARCHAR(20)
);

CREATE TABLE pokemon_type (
    id INT AUTO_INCREMENT PRIMARY KEY,
    pokemon_id INT NOT NULL,
    type_name VARCHAR(20) NOT NULL,
    FOREIGN KEY (pokemon_id) REFERENCES pokemon(id)
);

-- Jefes de raid
CREATE TABLE raid_boss (
    id INT AUTO_INCREMENT PRIMARY KEY,
    pokemon_id INT NOT NULL,
    tier VARCHAR(20),
    FOREIGN KEY (pokemon_id) REFERENCES pokemon(id)
);

-- Counters recomendados por jefe
CREATE TABLE counters (
    id INT AUTO_INCREMENT PRIMARY KEY,
    boss_id INT NOT NULL,
    counter_id INT NOT NULL,
    dps DECIMAL(5,2),
    recommendation VARCHAR(120),
    is_legendary BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (boss_id) REFERENCES raid_boss(id),
    FOREIGN KEY (counter_id) REFERENCES pokemon(id)
);

-- Eventos de Pokémon GO
CREATE TABLE events (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(120) NOT NULL,
    type VARCHAR(40) NOT NULL,
    startDate DATE NOT NULL,
    endDate DATE NOT NULL,
    description TEXT,
    tips TEXT
);

-- Pokémon del entrenador (para stardust planner con login)
CREATE TABLE trainer_pokemon (
    id INT AUTO_INCREMENT PRIMARY KEY,
    trainer_name VARCHAR(60),
    pokemon_id INT NOT NULL,
    cp INT,
    hp INT,
    stardust_cost INT,
    purpose VARCHAR(20),
    FOREIGN KEY (pokemon_id) REFERENCES pokemon(id)
);

-- Aplica estas alteraciones si la tabla equipos ya existe
ALTER TABLE equipos MODIFY autor VARCHAR(80) DEFAULT 'Delgado Cerros Rodrigo Daniel';
ALTER TABLE equipos ADD COLUMN IF NOT EXISTS notas TEXT;
